1. **Вхідні дані (токенізований текст)**: Процес починається з токенізованого тексту, який перетворюється на числові представлення.
2. **Шар токенних і позиційних embedding**: Токенізований текст проходить через шар **token embedding** і шар **positional embedding**, який фіксує позицію токенів у послідовності, що є критично важливим для розуміння порядку слів.
3. **Transformer blocks**: Модель містить **12 transformer blocks**, кожен із яких складається з кількох шарів. У цих блоках повторюється така послідовність:
- **Masked Multi-Head Attention**: Дає моделі змогу одночасно зосереджуватися на різних частинах вхідного тексту.
- **Layer Normalization**: Етап нормалізації для стабілізації та покращення навчання.
- **Feed Forward Layer**: Відповідає за обробку інформації від attention layer і прогнозування наступного токена.
- **Dropout Layers**: Ці шари запобігають перенавчанню, випадково відкидаючи units під час навчання.
4. **Фінальний вихідний шар**: Модель генерує **4x50,257-dimensional tensor**, де **50,257** позначає розмір словника. Кожен рядок цього tensor відповідає вектору, який модель використовує для прогнозування наступного слова в послідовності.
5. **Мета**: Мета полягає в тому, щоб перетворити ці embeddings назад на текст. Зокрема, останній рядок output використовується для генерації наступного слова, представленого як "forward" на цій схемі.

### Представлення коду
```python
import torch
import torch.nn as nn
import tiktoken

class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))

class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
return self.layers(x)

class MultiHeadAttention(nn.Module):
def __init__(self, d_in, d_out, context_length, dropout, num_heads, qkv_bias=False):
super().__init__()
assert d_out % num_heads == 0, "d_out must be divisible by num_heads"

self.d_out = d_out
self.num_heads = num_heads
self.head_dim = d_out // num_heads # Reduce the projection dim to match desired output dim

self.W_query = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_key = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_value = nn.Linear(d_in, d_out, bias=qkv_bias)
self.out_proj = nn.Linear(d_out, d_out)  # Linear layer to combine head outputs
self.dropout = nn.Dropout(dropout)
self.register_buffer('mask', torch.triu(torch.ones(context_length, context_length), diagonal=1))

def forward(self, x):
b, num_tokens, d_in = x.shape

keys = self.W_key(x) # Shape: (b, num_tokens, d_out)
queries = self.W_query(x)
values = self.W_value(x)

# We implicitly split the matrix by adding a `num_heads` dimension
# Unroll last dim: (b, num_tokens, d_out) -> (b, num_tokens, num_heads, head_dim)
keys = keys.view(b, num_tokens, self.num_heads, self.head_dim)
values = values.view(b, num_tokens, self.num_heads, self.head_dim)
queries = queries.view(b, num_tokens, self.num_heads, self.head_dim)

# Transpose: (b, num_tokens, num_heads, head_dim) -> (b, num_heads, num_tokens, head_dim)
keys = keys.transpose(1, 2)
queries = queries.transpose(1, 2)
values = values.transpose(1, 2)

# Compute scaled dot-product attention (aka self-attention) with a causal mask
attn_scores = queries @ keys.transpose(2, 3)  # Dot product for each head

# Original mask truncated to the number of tokens and converted to boolean
mask_bool = self.mask.bool()[:num_tokens, :num_tokens]

# Use the mask to fill attention scores
attn_scores.masked_fill_(mask_bool, -torch.inf)

attn_weights = torch.softmax(attn_scores / keys.shape[-1]**0.5, dim=-1)
attn_weights = self.dropout(attn_weights)

# Shape: (b, num_tokens, num_heads, head_dim)
context_vec = (attn_weights @ values).transpose(1, 2)

# Combine heads, where self.d_out = self.num_heads * self.head_dim
context_vec = context_vec.contiguous().view(b, num_tokens, self.d_out)
context_vec = self.out_proj(context_vec) # optional projection

return context_vec

class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"])
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# Shortcut connection for attention block
shortcut = x
x = self.norm1(x)
x = self.att(x)  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

# Shortcut connection for feed forward block
shortcut = x
x = self.norm2(x)
x = self.ff(x)
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

return x


class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])])

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(
cfg["emb_dim"], cfg["vocab_size"], bias=False
)

def forward(self, in_idx):
batch_size, seq_len = in_idx.shape
tok_embeds = self.tok_emb(in_idx)
pos_embeds = self.pos_emb(torch.arange(seq_len, device=in_idx.device))
x = tok_embeds + pos_embeds  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_emb(x)
x = self.trf_blocks(x)
x = self.final_norm(x)
logits = self.out_head(x)
return logits

GPT_CONFIG_124M = {
"vocab_size": 50257,    # Vocabulary size
"context_length": 1024, # Context length
"emb_dim": 768,         # Embedding dimension
"n_heads": 12,          # Number of attention heads
"n_layers": 12,         # Number of layers
"drop_rate": 0.1,       # Dropout rate
"qkv_bias": False       # Query-Key-Value bias
}

torch.manual_seed(123)
model = GPTModel(GPT_CONFIG_124M)
out = model(batch)
print("Input batch:\n", batch)
print("\nOutput shape:", out.shape)
print(out)
```

```python
def __init__(self):
super().__init__()

def forward(self, x):
```

#### **Призначення та функціональність**

- **GELU (Gaussian Error Linear Unit):** Функція активації, яка додає нелінійність у модель.
- **Плавна активація:** На відміну від ReLU, яка обнуляє від'ємні вхідні значення, GELU плавно перетворює входи на виходи, допускаючи малі ненульові значення для від'ємних входів.

### **Нейронна мережа FeedForward**

_Форми додано як коментарі для кращого розуміння форм матриць:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

x = self.layers[0](x)# x shape: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[1](x) # x shape remains: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[2](x) # x shape: (batch_size, seq_len, emb_dim)
return x  # Output shape: (batch_size, seq_len, emb_dim)
```
#### **Призначення та функціональність**

- **Позиційна FeedForward Network:** Застосовує двошарову повнозв'язну мережу до кожної позиції окремо та однаково.
- **Деталі шарів:**
- **Перший лінійний шар:** Розширює розмірність з `emb_dim` до `4 * emb_dim`.
- **Активація GELU:** Застосовує нелінійність.
- **Другий лінійний шар:** Зменшує розмірність назад до `emb_dim`.

> [!TIP]
> Як бачите, мережа Feed Forward використовує 3 шари. Перший з них — лінійний шар, який збільшує розмірність у 4 рази за допомогою лінійних ваг (параметрів, які навчаються всередині моделі). Потім функція GELU застосовується до всіх цих розмірностей для внесення нелінійних варіацій, щоб отримати більш насичені представлення, і нарешті використовується ще один лінійний шар, щоб повернутися до початкового розміру.

### **Механізм багатоголової уваги**

Це вже було пояснено в попередньому розділі.

#### **Призначення та функціональність**

- **Багатоголова Self-Attention:** Дозволяє моделі зосереджуватися на різних позиціях у вхідній послідовності під час кодування токена.
- **Ключові компоненти:**
- **Queries, Keys, Values:** Лінійні проєкції входу, які використовуються для обчислення оцінок уваги.
- **Heads:** Кілька механізмів уваги, що працюють паралельно (`num_heads`), кожен зі зменшеною розмірністю (`head_dim`).
- **Оцінки уваги:** Обчислюються як скалярний добуток queries і keys, масштабуються та маскуються.
- **Маскування:** Застосовується causal mask, щоб запобігти звертанню моделі до майбутніх токенів (це важливо для autoregressive моделей, таких як GPT).
- **Ваги уваги:** Softmax замаскованих і масштабованих оцінок уваги.
- **Контекстний вектор:** Зважена сума values відповідно до ваг уваги.
- **Вихідна проєкція:** Лінійний шар для об'єднання виходів усіх heads.

> [!TIP]
> Мета цієї мережі — знаходити зв'язки між токенами в одному контексті. Крім того, токени розподіляються між різними heads, щоб запобігти перенавчанню, хоча остаточні зв'язки, знайдені кожною head, наприкінці цієї мережі об'єднуються.
>
> Крім того, під час навчання застосовується **causal mask**, щоб пізніші токени не бралися до уваги під час пошуку конкретних зв'язків із токеном, а також застосовується певний **dropout**, щоб **запобігти перенавчанню**.

### **Нормалізація** шару
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5 # Prevent division by zero during normalization.
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift
```
#### **Призначення та функціональність**

- **Нормалізація шарів:** Техніка, яка використовується для нормалізації вхідних даних за ознаками (вимірами embedding) для кожного окремого прикладу в batch.
- **Компоненти:**
- **`eps`:** Невелика константа (`1e-5`), що додається до дисперсії для запобігання діленню на нуль під час нормалізації.
- **`scale` і `shift`:** Параметри, що навчаються (`nn.Parameter`), які дають змогу моделі масштабувати та зміщувати нормалізований результат. Вони відповідно ініціалізуються одиницями та нулями.
- **Процес нормалізації:**
- **Обчислення середнього (`mean`):** Обчислює середнє значення вхідного `x` за виміром embedding (`dim=-1`), зберігаючи цей вимір для broadcasting (`keepdim=True`).
- **Обчислення дисперсії (`var`):** Обчислює дисперсію `x` за виміром embedding, також зберігаючи цей вимір. Параметр `unbiased=False` гарантує, що дисперсія обчислюється за допомогою зміщеної оцінки (ділення на `N`, а не на `N-1`), що є доречним під час нормалізації за ознаками, а не за вибірками.
- **Нормалізація (`norm_x`):** Віднімає середнє значення від `x` і ділить результат на квадратний корінь із дисперсії плюс `eps`.
- **Масштабування та зміщення:** Застосовує параметри `scale` і `shift`, що навчаються, до нормалізованого результату.

> [!TIP]
> Мета полягає в тому, щоб забезпечити середнє значення 0 і дисперсію 1 за всіма вимірами одного й того самого token. Це потрібно для **стабілізації навчання глибоких нейронних мереж** шляхом зменшення внутрішнього коваріатного зсуву, який означає зміну розподілу активацій мережі через оновлення параметрів під час навчання.

### **Transformer Block**

_До коментарів додано форми, щоб краще зрозуміти форми матриць:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"]
)
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for attention block
shortcut = x  # shape: (batch_size, seq_len, emb_dim)
x = self.norm1(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.att(x)    # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for feedforward block
shortcut = x       # shape: (batch_size, seq_len, emb_dim)
x = self.norm2(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.ff(x)     # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

return x  # Output shape: (batch_size, seq_len, emb_dim)

```
#### **Призначення та функціональність**

- **Композиція шарів:** Поєднує multi-head attention, feedforward network, layer normalization і residual connections.
- **Layer Normalization:** Застосовується перед attention- і feedforward-шарами для стабільного навчання.
- **Residual Connections (Shortcuts):** Додають вхід шару до його виходу, щоб покращити проходження градієнта та забезпечити навчання глибоких мереж.
- **Dropout:** Застосовується після attention- і feedforward-шарів для регуляризації.

#### **Покрокова функціональність**

1. **First Residual Path (Self-Attention):**
- **Input (`shortcut`):** Зберегти оригінальний вхід для residual connection.
- **Layer Norm (`norm1`):** Нормалізувати вхід.
- **Multi-Head Attention (`att`):** Застосувати self-attention.
- **Dropout (`drop_shortcut`):** Застосувати dropout для регуляризації.
- **Add Residual (`x + shortcut`):** Об'єднати з оригінальним входом.
2. **Second Residual Path (FeedForward):**
- **Input (`shortcut`):** Зберегти оновлений вхід для наступної residual connection.
- **Layer Norm (`norm2`):** Нормалізувати вхід.
- **FeedForward Network (`ff`):** Застосувати feedforward-перетворення.
- **Dropout (`drop_shortcut`):** Застосувати dropout.
- **Add Residual (`x + shortcut`):** Об'єднати з входом із першого residual path.

> [!TIP]
> Transformer block об'єднує всі мережі та застосовує певні **нормалізацію** і **dropouts**, щоб покращити стабільність навчання та результати.\
> Зверніть увагу, що dropouts виконуються після використання кожної мережі, тоді як нормалізація застосовується перед нею.
>
> Крім того, він використовує shortcuts, які полягають у **додаванні виходу мережі до її входу**. Це допомагає запобігти проблемі зникнення градієнта, забезпечуючи внесок початкових шарів "такою ж мірою", як і останніх.

### **GPTModel**

_До коментарів додано розміри, щоб краще зрозуміти розміри матриць:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
# shape: (vocab_size, emb_dim)

self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
# shape: (context_length, emb_dim)

self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])]
)
# Stack of TransformerBlocks

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(cfg["emb_dim"], cfg["vocab_size"], bias=False)
# shape: (emb_dim, vocab_size)

def forward(self, in_idx):
# in_idx shape: (batch_size, seq_len)
batch_size, seq_len = in_idx.shape

# Token embeddings
tok_embeds = self.tok_emb(in_idx)
# shape: (batch_size, seq_len, emb_dim)

# Positional embeddings
pos_indices = torch.arange(seq_len, device=in_idx.device)
# shape: (seq_len,)
pos_embeds = self.pos_emb(pos_indices)
# shape: (seq_len, emb_dim)

# Add token and positional embeddings
x = tok_embeds + pos_embeds  # Broadcasting over batch dimension
# x shape: (batch_size, seq_len, emb_dim)

x = self.drop_emb(x)  # Dropout applied
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.trf_blocks(x)  # Pass through Transformer blocks
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.final_norm(x)  # Final LayerNorm
# x shape remains: (batch_size, seq_len, emb_dim)

logits = self.out_head(x)  # Project to vocabulary size
# logits shape: (batch_size, seq_len, vocab_size)

return logits  # Output shape: (batch_size, seq_len, vocab_size)
```
#### **Призначення та функціональність**

- **Embedding Layers:**
- **Token Embeddings (`tok_emb`):** Перетворює індекси токенів на embedding-и. Нагадаємо, це ваги, призначені кожному виміру кожного токена у словнику.
- **Positional Embeddings (`pos_emb`):** Додає до embedding-ів позиційну інформацію, щоб враховувати порядок токенів. Нагадаємо, це ваги, призначені токену відповідно до його позиції в тексті.
- **Dropout (`drop_emb`):** Застосовується до embedding-ів для regularisation.
- **Transformer Blocks (`trf_blocks`):** Стек із `n_layers` transformer blocks для обробки embedding-ів.
- **Final Normalization (`final_norm`):** Layer normalization перед вихідним шаром.
- **Output Layer (`out_head`):** Проєктує фінальні hidden states на розмір словника, щоб отримати logits для prediction.

> [!TIP]
> Мета цього класу — використовувати всі інші згадані мережі, щоб **передбачати наступний токен у послідовності**, що є фундаментальним для таких задач, як генерація тексту.
>
> Зверніть увагу, що він **використовуватиме стільки transformer blocks, скільки вказано**, а кожен transformer block використовує одну multi-head attestation net, одну feed forward net і кілька normalizations. Отже, якщо використовується 12 transformer blocks, помножте це на 12.
>
> Крім того, **normalization** layer додається **перед** **output**, а в кінці застосовується final linear layer, щоб отримати результати з правильними розмірами. Зверніть увагу, що кожен фінальний vector має розмір використаного словника. Це пояснюється тим, що модель намагається отримати ймовірність для кожного можливого токена у словнику.

### **Покроковий розрахунок**

#### **1. Шари embedding: Token Embedding і Position Embedding**

- **Шар:** `nn.Embedding(vocab_size, emb_dim)`
- **Параметри:** `vocab_size * emb_dim`
```python
token_embedding_params = 50257 * 768 = 38,597,376
```
- **Шар:** `nn.Embedding(context_length, emb_dim)`
- **Параметри:** `context_length * emb_dim`
```python
position_embedding_params = 1024 * 768 = 786,432
```
**Загальна кількість параметрів embedding**
```python
embedding_params = token_embedding_params + position_embedding_params
embedding_params = 38,597,376 + 786,432 = 39,383,808
```
#### **2. Трансформерні блоки**

Є 12 трансформерних блоків, тому обчислимо кількість параметрів для одного блока, а потім помножимо на 12.

**Параметри одного трансформерного блока**

**a. Багатоголовкова увага**

- **Компоненти:**
- **Лінійний шар Query (`W_query`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Лінійний шар Key (`W_key`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Лінійний шар Value (`W_value`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Вихідна проєкція (`out_proj`):** `nn.Linear(emb_dim, emb_dim)`
- **Обчислення:**

- **Кожен із `W_query`, `W_key`, `W_value`:**

```python
qkv_params = emb_dim * emb_dim = 768 * 768 = 589,824
```

Оскільки таких шарів три:

```python
total_qkv_params = 3 * qkv_params = 3 * 589,824 = 1,769,472
```

- **Вихідна проєкція (`out_proj`):**

```python
out_proj_params = (emb_dim * emb_dim) + emb_dim = (768 * 768) + 768 = 589,824 + 768 = 590,592
```

- **Загальна кількість параметрів багатоголовкової уваги:**

```python
mha_params = total_qkv_params + out_proj_params
```

- **Компоненти:**

## Генерування тексту

Якщо модель прогнозує наступний токен, як у попередньому прикладі, потрібно лише взяти значення останнього токена з вихідних даних (оскільки саме вони відповідатимуть прогнозованому токену). Це буде **значення для кожного елемента словника**. Потім потрібно використати функцію `softmax`, щоб нормалізувати виміри до ймовірностей, сума яких дорівнює 1, і визначити індекс найбільшого елемента, який буде індексом слова у словнику.

Код із [https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb](https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb):<sup>[[1]](#references)</sup>
```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
# idx is (batch, n_tokens) array of indices in the current context
for _ in range(max_new_tokens):

# Crop current context if it exceeds the supported context size
# E.g., if LLM supports only 5 tokens, and the context size is 10
# then only the last 5 tokens are used as context
idx_cond = idx[:, -context_size:]

# Get the predictions
with torch.no_grad():
logits = model(idx_cond)

# Focus only on the last time step
# (batch, n_tokens, vocab_size) becomes (batch, vocab_size)
logits = logits[:, -1, :]

# Apply softmax to get probabilities
probas = torch.softmax(logits, dim=-1)  # (batch, vocab_size)

# Get the idx of the vocab entry with the highest probability value
idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # (batch, 1)

# Append sampled index to the running sequence
idx = torch.cat((idx, idx_next), dim=1)  # (batch, n_tokens+1)

return idx


start_context = "Hello, I am"

encoded = tokenizer.encode(start_context)
print("encoded:", encoded)

encoded_tensor = torch.tensor(encoded).unsqueeze(0)
print("encoded_tensor.shape:", encoded_tensor.shape)

model.eval() # disable dropout

out = generate_text_simple(
model=model,
idx=encoded_tensor,
max_new_tokens=6,
context_size=GPT_CONFIG_124M["context_length"]
)

print("Output:", out)
print("Output length:", len(out[0]))
```
