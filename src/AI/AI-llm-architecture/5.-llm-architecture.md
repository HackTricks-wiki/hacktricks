1. **입력(Tokenized Text)**: 프로세스는 tokenized text로 시작하며, 이는 수치 표현으로 변환됩니다.
2. **Token Embedding 및 Positional Embedding Layer**: tokenized text는 **token embedding** 레이어와 **positional embedding layer**를 통과합니다. positional embedding layer는 시퀀스에서 토큰의 위치를 포착하며, 단어 순서를 이해하는 데 중요합니다.
3. **Transformer Blocks**: 모델에는 각각 여러 레이어로 구성된 **12개의 transformer blocks**가 포함됩니다. 이 블록에서는 다음 시퀀스가 반복됩니다:
- **Masked Multi-Head Attention**: 모델이 입력 텍스트의 서로 다른 부분에 동시에 집중할 수 있도록 합니다.
- **Layer Normalization**: training을 안정화하고 개선하기 위한 정규화 단계입니다.
- **Feed Forward Layer**: attention layer의 정보를 처리하고 다음 토큰을 예측하는 역할을 합니다.
- **Dropout Layers**: training 중 유닛을 무작위로 제거하여 overfitting을 방지합니다.
4. **Final Output Layer**: 모델은 **4x50,257차원 tensor**를 출력하며, 여기서 **50,257**은 vocabulary의 크기를 나타냅니다. 이 tensor의 각 행은 모델이 시퀀스에서 다음 단어를 예측하는 데 사용하는 벡터에 해당합니다.
5. **목표**: 이러한 embedding을 다시 텍스트로 변환하는 것이 목표입니다. 구체적으로는 output의 마지막 행을 사용하여 다음 단어를 생성하며, 이 다이어그램에서는 "forward"로 표시됩니다.

### Code representation
```python
import torch
import torch.nn as nn
import tiktoken

class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))

class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
return self.layers(x)

class MultiHeadAttention(nn.Module):
def __init__(self, d_in, d_out, context_length, dropout, num_heads, qkv_bias=False):
super().__init__()
assert d_out % num_heads == 0, "d_out must be divisible by num_heads"

self.d_out = d_out
self.num_heads = num_heads
self.head_dim = d_out // num_heads # Reduce the projection dim to match desired output dim

self.W_query = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_key = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_value = nn.Linear(d_in, d_out, bias=qkv_bias)
self.out_proj = nn.Linear(d_out, d_out)  # Linear layer to combine head outputs
self.dropout = nn.Dropout(dropout)
self.register_buffer('mask', torch.triu(torch.ones(context_length, context_length), diagonal=1))

def forward(self, x):
b, num_tokens, d_in = x.shape

keys = self.W_key(x) # Shape: (b, num_tokens, d_out)
queries = self.W_query(x)
values = self.W_value(x)

# We implicitly split the matrix by adding a `num_heads` dimension
# Unroll last dim: (b, num_tokens, d_out) -> (b, num_tokens, num_heads, head_dim)
keys = keys.view(b, num_tokens, self.num_heads, self.head_dim)
values = values.view(b, num_tokens, self.num_heads, self.head_dim)
queries = queries.view(b, num_tokens, self.num_heads, self.head_dim)

# Transpose: (b, num_tokens, num_heads, head_dim) -> (b, num_heads, num_tokens, head_dim)
keys = keys.transpose(1, 2)
queries = queries.transpose(1, 2)
values = values.transpose(1, 2)

# Compute scaled dot-product attention (aka self-attention) with a causal mask
attn_scores = queries @ keys.transpose(2, 3)  # Dot product for each head

# Original mask truncated to the number of tokens and converted to boolean
mask_bool = self.mask.bool()[:num_tokens, :num_tokens]

# Use the mask to fill attention scores
attn_scores.masked_fill_(mask_bool, -torch.inf)

attn_weights = torch.softmax(attn_scores / keys.shape[-1]**0.5, dim=-1)
attn_weights = self.dropout(attn_weights)

# Shape: (b, num_tokens, num_heads, head_dim)
context_vec = (attn_weights @ values).transpose(1, 2)

# Combine heads, where self.d_out = self.num_heads * self.head_dim
context_vec = context_vec.contiguous().view(b, num_tokens, self.d_out)
context_vec = self.out_proj(context_vec) # optional projection

return context_vec

class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"])
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# Shortcut connection for attention block
shortcut = x
x = self.norm1(x)
x = self.att(x)  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

# Shortcut connection for feed forward block
shortcut = x
x = self.norm2(x)
x = self.ff(x)
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

return x


class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])])

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(
cfg["emb_dim"], cfg["vocab_size"], bias=False
)

def forward(self, in_idx):
batch_size, seq_len = in_idx.shape
tok_embeds = self.tok_emb(in_idx)
pos_embeds = self.pos_emb(torch.arange(seq_len, device=in_idx.device))
x = tok_embeds + pos_embeds  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_emb(x)
x = self.trf_blocks(x)
x = self.final_norm(x)
logits = self.out_head(x)
return logits

GPT_CONFIG_124M = {
"vocab_size": 50257,    # Vocabulary size
"context_length": 1024, # Context length
"emb_dim": 768,         # Embedding dimension
"n_heads": 12,          # Number of attention heads
"n_layers": 12,         # Number of layers
"drop_rate": 0.1,       # Dropout rate
"qkv_bias": False       # Query-Key-Value bias
}

torch.manual_seed(123)
model = GPTModel(GPT_CONFIG_124M)
out = model(batch)
print("Input batch:\n", batch)
print("\nOutput shape:", out.shape)
print(out)
```

### **GELU 활성화 함수**
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))
```
#### **목적 및 기능**

- **GELU (Gaussian Error Linear Unit):** 모델에 비선형성을 추가하는 activation function입니다.
- **Smooth Activation:** 음수 입력을 0으로 만드는 ReLU와 달리, GELU는 입력을 출력으로 부드럽게 매핑하여 음수 입력에도 작지만 0이 아닌 값을 허용합니다.

### **FeedForward Neural Network**

_행렬의 shape를 더 쉽게 이해할 수 있도록 주석으로 추가했습니다:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

x = self.layers[0](x)# x shape: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[1](x) # x shape remains: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[2](x) # x shape: (batch_size, seq_len, emb_dim)
return x  # Output shape: (batch_size, seq_len, emb_dim)
```
#### **목적 및 기능**

- **Position-wise FeedForward Network:** 각 position에 동일하게 두 개의 fully connected layer를 적용합니다.
- **Layer 세부 사항:**
- **First Linear Layer:** 차원을 `emb_dim`에서 `4 * emb_dim`으로 확장합니다.
- **GELU Activation:** 비선형성을 적용합니다.
- **Second Linear Layer:** 차원을 다시 `emb_dim`으로 축소합니다.

> [!TIP]
> 보시다시피 Feed Forward network는 3개의 layer를 사용합니다. 첫 번째는 linear layer로, linear weights(모델 내부에서 학습되는 parameters)를 사용해 차원을 4배로 확장합니다. 그런 다음 모든 차원에 GELU function을 적용하여 비선형 변화를 만들고 더 풍부한 representation을 포착합니다. 마지막으로 또 다른 linear layer를 사용해 차원을 원래 크기로 되돌립니다.

### **Multi-Head Attention Mechanism**

이는 이전 section에서 이미 설명했습니다.

#### **목적 및 기능**

- **Multi-Head Self-Attention:** 모델이 token을 encoding할 때 input sequence 내 서로 다른 position에 집중할 수 있도록 합니다.
- **주요 구성 요소:**
- **Queries, Keys, Values:** attention scores를 계산하는 데 사용되는 input의 linear projections입니다.
- **Heads:** 병렬로 실행되는 여러 attention mechanisms(`num_heads`)이며, 각각 축소된 차원(`head_dim`)을 사용합니다.
- **Attention Scores:** queries와 keys의 dot product로 계산되며, scaling 및 masking이 적용됩니다.
- **Masking:** 모델이 future tokens에 attention하는 것을 방지하기 위해 causal mask가 적용됩니다(GPT와 같은 autoregressive models에서 중요합니다).
- **Attention Weights:** masking 및 scaling된 attention scores에 Softmax를 적용한 결과입니다.
- **Context Vector:** attention weights에 따른 values의 weighted sum입니다.
- **Output Projection:** 모든 heads의 outputs를 결합하는 linear layer입니다.

> [!TIP]
> 이 network의 목표는 동일한 context에 있는 tokens 간의 relations를 찾는 것입니다. 또한 overfitting을 방지하기 위해 tokens는 서로 다른 heads로 나뉘지만, 각 head에서 찾아낸 최종 relations는 이 network의 마지막에서 결합됩니다.
>
> 또한 training 중에는 특정 token과의 관계를 확인할 때 뒤에 있는 tokens가 고려되지 않도록 **causal mask**가 적용되며, **overfitting을 방지하기 위해** 일부 **dropout**도 적용됩니다.

### **Layer** Normalization
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5 # Prevent division by zero during normalization.
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift
```
#### **목적 및 기능**

- **Layer Normalization:** 배치의 각 개별 예제에 대해 feature(embedding 차원) 전체의 입력을 정규화하는 데 사용되는 기법입니다.
- **구성 요소:**
- **`eps`:** 정규화 중 0으로 나누는 것을 방지하기 위해 variance에 더하는 작은 상수(`1e-5`)입니다.
- **`scale` 및 `shift`:** 정규화된 출력을 scaling하고 shifting할 수 있도록 하는 학습 가능한 파라미터(`nn.Parameter`)입니다. 각각 1과 0으로 초기화됩니다.
- **정규화 과정:**
- **평균 계산 (`mean`):** embedding 차원(`dim=-1)`)을 기준으로 입력 `x`의 평균을 계산하며, broadcasting을 위해 차원을 유지합니다(`keepdim=True`).
- **분산 계산 (`var`):** embedding 차원을 기준으로 `x`의 variance를 계산하며, 이때도 차원을 유지합니다. `unbiased=False` 파라미터는 biased estimator를 사용해 variance를 계산하도록 합니다. 즉, `N-1`이 아닌 `N`으로 나누며, 이는 samples가 아닌 features를 기준으로 정규화할 때 적절합니다.
- **정규화 (`norm_x`):** `x`에서 평균을 빼고, variance에 `eps`를 더한 값의 제곱근으로 나눕니다.
- **Scaling 및 Shifting:** 학습 가능한 `scale` 및 `shift` 파라미터를 정규화된 출력에 적용합니다.

> [!TIP]
> 목표는 동일한 token의 모든 차원에 걸쳐 평균을 0, variance를 1로 만드는 것입니다. 이는 **deep neural networks의 training을 안정화**하기 위한 것으로, training 중 파라미터가 업데이트되면서 network activations의 분포가 변하는 현상인 internal covariate shift를 줄입니다.

### **Transformer Block**

_행렬의 shape를 더 잘 이해할 수 있도록 주석으로 추가했습니다:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"]
)
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for attention block
shortcut = x  # shape: (batch_size, seq_len, emb_dim)
x = self.norm1(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.att(x)    # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for feedforward block
shortcut = x       # shape: (batch_size, seq_len, emb_dim)
x = self.norm2(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.ff(x)     # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

return x  # Output shape: (batch_size, seq_len, emb_dim)

```
#### **목적 및 기능**

- **레이어 구성:** multi-head attention, feedforward network, layer normalization, residual connection을 결합합니다.
- **Layer Normalization:** 안정적인 training을 위해 attention 및 feedforward layer 전에 적용합니다.
- **Residual Connections (Shortcuts):** gradient flow를 개선하고 deep network의 training을 가능하게 하기 위해 layer의 입력을 출력에 더합니다.
- **Dropout:** regularization을 위해 attention 및 feedforward layer 후에 적용합니다.

#### **단계별 기능**

1. **첫 번째 Residual Path (Self-Attention):**
- **입력 (`shortcut`):** residual connection을 위해 원래 입력을 저장합니다.
- **Layer Norm (`norm1`):** 입력을 normalize합니다.
- **Multi-Head Attention (`att`):** self-attention을 적용합니다.
- **Dropout (`drop_shortcut`):** regularization을 위해 dropout을 적용합니다.
- **Residual 추가 (`x + shortcut`):** 원래 입력과 결합합니다.
2. **두 번째 Residual Path (FeedForward):**
- **입력 (`shortcut`):** 다음 residual connection을 위해 업데이트된 입력을 저장합니다.
- **Layer Norm (`norm2`):** 입력을 normalize합니다.
- **FeedForward Network (`ff`):** feedforward transformation을 적용합니다.
- **Dropout (`drop_shortcut`):** dropout을 적용합니다.
- **Residual 추가 (`x + shortcut`):** 첫 번째 residual path의 입력과 결합합니다.

> [!TIP]
> transformer block은 모든 network를 함께 묶고, training stability와 결과를 개선하기 위해 **normalization** 및 **dropout**을 적용합니다.\
> 각 network를 사용한 후에 dropout이 수행되고, normalization은 그 전에 적용된다는 점에 유의하세요.
>
> 또한 **network의 출력과 입력을 더하는 것**으로 구성된 shortcut을 사용합니다. 이를 통해 초기 layer가 마지막 layer만큼 "많이" 기여하도록 하여 vanishing gradient 문제를 방지할 수 있습니다.

```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
# shape: (vocab_size, emb_dim)

self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
# shape: (context_length, emb_dim)

self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])]
)
# Stack of TransformerBlocks

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(cfg["emb_dim"], cfg["vocab_size"], bias=False)
# shape: (emb_dim, vocab_size)

def forward(self, in_idx):
# in_idx shape: (batch_size, seq_len)
batch_size, seq_len = in_idx.shape

# Token embeddings
tok_embeds = self.tok_emb(in_idx)
# shape: (batch_size, seq_len, emb_dim)

# Positional embeddings
pos_indices = torch.arange(seq_len, device=in_idx.device)
# shape: (seq_len,)
pos_embeds = self.pos_emb(pos_indices)
# shape: (seq_len, emb_dim)

# Add token and positional embeddings
x = tok_embeds + pos_embeds  # Broadcasting over batch dimension
# x shape: (batch_size, seq_len, emb_dim)

x = self.drop_emb(x)  # Dropout applied
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.trf_blocks(x)  # Pass through Transformer blocks
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.final_norm(x)  # Final LayerNorm
# x shape remains: (batch_size, seq_len, emb_dim)

logits = self.out_head(x)  # Project to vocabulary size
# logits shape: (batch_size, seq_len, vocab_size)

return logits  # Output shape: (batch_size, seq_len, vocab_size)
```
#### **목적 및 기능**

- **Embedding Layers:**
- **Token Embeddings (`tok_emb`):** 토큰 인덱스를 임베딩으로 변환합니다. 다시 말해, 이는 vocabulary의 각 토큰에 있는 각 차원에 부여된 가중치입니다.
- **Positional Embeddings (`pos_emb`):** 토큰의 순서를 반영하기 위해 임베딩에 위치 정보를 추가합니다. 다시 말해, 이는 텍스트에서 토큰의 위치에 따라 토큰에 부여되는 가중치입니다.
- **Dropout (`drop_emb`):** 정규화를 위해 임베딩에 적용됩니다.
- **Transformer Blocks (`trf_blocks`):** 임베딩을 처리하는 `n_layers`개의 transformer block 스택입니다.
- **Final Normalization (`final_norm`):** output layer 전에 적용되는 layer normalization입니다.
- **Output Layer (`out_head`):** 최종 hidden states를 vocabulary 크기로 projection하여 예측을 위한 logits를 생성합니다.

> [!TIP]
> 이 class의 목표는 앞서 언급한 다른 모든 network를 사용하여 **sequence에서 다음 token을 예측하는 것**이며, 이는 text generation과 같은 작업의 기본이 됩니다.
>
> 이 class가 **지정된 수만큼 transformer block을 사용한다는 점**에 유의하세요. 또한 각 transformer block은 하나의 multi-head attestation net, 하나의 feed forward net 및 여러 개의 normalization을 사용합니다. 따라서 12개의 transformer block을 사용한다면 이 구성을 12배로 계산해야 합니다.
>
> 또한 **output** 전에 **normalization** layer가 추가되고, 마지막에 최종 linear layer가 적용되어 올바른 차원의 결과를 얻습니다. 각 최종 vector의 크기가 사용된 vocabulary의 크기와 같다는 점에 유의하세요. 이는 vocabulary 내에서 가능한 각 token에 대한 probability를 구하려고 하기 때문입니다.

#### **1. Embedding Layers: Token Embedding & Position Embedding**

- **Layer:** `nn.Embedding(vocab_size, emb_dim)`
- **Parameters:** `vocab_size * emb_dim`

- **층:** `nn.Embedding(context_length, emb_dim)`

#### **2. Transformer Blocks**

12개의 transformer block이 있으므로, 하나의 block에 대한 parameter를 계산한 다음 12를 곱합니다.

**a. Multi-Head Attention**

- **Value Linear Layer (`W_value`):** `nn.Linear(emb_dim, emb_dim, bias=False)`

## Text 생성

이전 토큰과 같은 방식으로 다음 token을 예측하는 model이 있다면, output에서 마지막 token의 값만 가져오면 된다(해당 값이 예측된 token의 값이기 때문). 이 값은 **vocabulary의 각 entry에 대한 값**이며, 그런 다음 `softmax` 함수를 사용해 차원을 합이 1인 probability로 normalize하고, 가장 큰 entry의 index를 가져오면 된다. 이 index가 vocabulary 내 단어의 index가 된다.

Code from [https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb](https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb):<sup>[[1]](#references)</sup>
```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
# idx is (batch, n_tokens) array of indices in the current context
for _ in range(max_new_tokens):

# Crop current context if it exceeds the supported context size
# E.g., if LLM supports only 5 tokens, and the context size is 10
# then only the last 5 tokens are used as context
idx_cond = idx[:, -context_size:]

# Get the predictions
with torch.no_grad():
logits = model(idx_cond)

# Focus only on the last time step
# (batch, n_tokens, vocab_size) becomes (batch, vocab_size)
logits = logits[:, -1, :]

# Apply softmax to get probabilities
probas = torch.softmax(logits, dim=-1)  # (batch, vocab_size)

# Get the idx of the vocab entry with the highest probability value
idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # (batch, 1)

# Append sampled index to the running sequence
idx = torch.cat((idx, idx_next), dim=1)  # (batch, n_tokens+1)

return idx


start_context = "Hello, I am"

encoded = tokenizer.encode(start_context)
print("encoded:", encoded)

encoded_tensor = torch.tensor(encoded).unsqueeze(0)
print("encoded_tensor.shape:", encoded_tensor.shape)

model.eval() # disable dropout

out = generate_text_simple(
model=model,
idx=encoded_tensor,
max_new_tokens=6,
context_size=GPT_CONFIG_124M["context_length"]
)

print("Output:", out)
print("Output length:", len(out[0]))
```
