1. **Input (Tokenized Text)**: प्रक्रिया tokenized text से शुरू होती है, जिसे numerical representations में बदला जाता है।
2. **Token Embedding and Positional Embedding Layer**: tokenized text को **token embedding** layer और **positional embedding layer** से गुजारा जाता है। यह sequence में tokens की position को capture करता है, जो word order को समझने के लिए महत्वपूर्ण है।
3. **Transformer Blocks**: model में **12 transformer blocks** होते हैं, जिनमें प्रत्येक में कई layers होती हैं। ये blocks निम्नलिखित sequence को दोहराते हैं:
- **Masked Multi-Head Attention**: model को input text के अलग-अलग हिस्सों पर एक साथ focus करने की अनुमति देता है।
- **Layer Normalization**: training को stabilize और बेहतर करने के लिए normalization का एक चरण।
- **Feed Forward Layer**: attention layer से प्राप्त information को process करने और अगले token के बारे में predictions करने के लिए जिम्मेदार।
- **Dropout Layers**: training के दौरान units को randomly drop करके ये layers overfitting को रोकती हैं।
4. **Final Output Layer**: model एक **4x50,257-dimensional tensor** output करता है, जहां **50,257** vocabulary के size को दर्शाता है। इस tensor की प्रत्येक row एक vector से संबंधित होती है, जिसका उपयोग model sequence में अगले word की prediction करने के लिए करता है।
5. **Goal**: उद्देश्य इन embeddings को वापस text में बदलना है। विशेष रूप से, output की last row का उपयोग अगले word को generate करने के लिए किया जाता है, जिसे इस diagram में "forward" के रूप में दर्शाया गया है।

### Code representation
```python
import torch
import torch.nn as nn
import tiktoken

class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))

class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
return self.layers(x)

class MultiHeadAttention(nn.Module):
def __init__(self, d_in, d_out, context_length, dropout, num_heads, qkv_bias=False):
super().__init__()
assert d_out % num_heads == 0, "d_out must be divisible by num_heads"

self.d_out = d_out
self.num_heads = num_heads
self.head_dim = d_out // num_heads # Reduce the projection dim to match desired output dim

self.W_query = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_key = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_value = nn.Linear(d_in, d_out, bias=qkv_bias)
self.out_proj = nn.Linear(d_out, d_out)  # Linear layer to combine head outputs
self.dropout = nn.Dropout(dropout)
self.register_buffer('mask', torch.triu(torch.ones(context_length, context_length), diagonal=1))

def forward(self, x):
b, num_tokens, d_in = x.shape

keys = self.W_key(x) # Shape: (b, num_tokens, d_out)
queries = self.W_query(x)
values = self.W_value(x)

# We implicitly split the matrix by adding a `num_heads` dimension
# Unroll last dim: (b, num_tokens, d_out) -> (b, num_tokens, num_heads, head_dim)
keys = keys.view(b, num_tokens, self.num_heads, self.head_dim)
values = values.view(b, num_tokens, self.num_heads, self.head_dim)
queries = queries.view(b, num_tokens, self.num_heads, self.head_dim)

# Transpose: (b, num_tokens, num_heads, head_dim) -> (b, num_heads, num_tokens, head_dim)
keys = keys.transpose(1, 2)
queries = queries.transpose(1, 2)
values = values.transpose(1, 2)

# Compute scaled dot-product attention (aka self-attention) with a causal mask
attn_scores = queries @ keys.transpose(2, 3)  # Dot product for each head

# Original mask truncated to the number of tokens and converted to boolean
mask_bool = self.mask.bool()[:num_tokens, :num_tokens]

# Use the mask to fill attention scores
attn_scores.masked_fill_(mask_bool, -torch.inf)

attn_weights = torch.softmax(attn_scores / keys.shape[-1]**0.5, dim=-1)
attn_weights = self.dropout(attn_weights)

# Shape: (b, num_tokens, num_heads, head_dim)
context_vec = (attn_weights @ values).transpose(1, 2)

# Combine heads, where self.d_out = self.num_heads * self.head_dim
context_vec = context_vec.contiguous().view(b, num_tokens, self.d_out)
context_vec = self.out_proj(context_vec) # optional projection

return context_vec

class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"])
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# Shortcut connection for attention block
shortcut = x
x = self.norm1(x)
x = self.att(x)  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

# Shortcut connection for feed forward block
shortcut = x
x = self.norm2(x)
x = self.ff(x)
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

return x


class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])])

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(
cfg["emb_dim"], cfg["vocab_size"], bias=False
)

def forward(self, in_idx):
batch_size, seq_len = in_idx.shape
tok_embeds = self.tok_emb(in_idx)
pos_embeds = self.pos_emb(torch.arange(seq_len, device=in_idx.device))
x = tok_embeds + pos_embeds  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_emb(x)
x = self.trf_blocks(x)
x = self.final_norm(x)
logits = self.out_head(x)
return logits

GPT_CONFIG_124M = {
"vocab_size": 50257,    # Vocabulary size
"context_length": 1024, # Context length
"emb_dim": 768,         # Embedding dimension
"n_heads": 12,          # Number of attention heads
"n_layers": 12,         # Number of layers
"drop_rate": 0.1,       # Dropout rate
"qkv_bias": False       # Query-Key-Value bias
}

torch.manual_seed(123)
model = GPTModel(GPT_CONFIG_124M)
out = model(batch)
print("Input batch:\n", batch)
print("\nOutput shape:", out.shape)
print(out)
```

#### **उद्देश्य और कार्यक्षमता**

- **GELU (Gaussian Error Linear Unit):** एक activation function जो model में non-linearity शामिल करता है।
- **Smooth Activation:** ReLU के विपरीत, जो negative inputs को zero कर देता है, GELU inputs को outputs में smoothly map करता है, जिससे negative inputs के लिए छोटे, non-zero values की अनुमति मिलती है।

### **FeedForward Neural Network**

_Matrices के shapes को बेहतर ढंग से समझने के लिए comments के रूप में shapes जोड़े गए हैं:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

x = self.layers[0](x)# x shape: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[1](x) # x shape remains: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[2](x) # x shape: (batch_size, seq_len, emb_dim)
return x  # Output shape: (batch_size, seq_len, emb_dim)
```
#### **उद्देश्य और कार्यक्षमता**

- **Position-wise FeedForward Network:** प्रत्येक position पर अलग-अलग और समान रूप से दो-स्तरीय fully connected network लागू करता है।
- **Layer Details:**
- **First Linear Layer:** dimensionality को `emb_dim` से `4 * emb_dim` तक बढ़ाता है।
- **GELU Activation:** non-linearity लागू करता है।
- **Second Linear Layer:** dimensionality को फिर से `emb_dim` तक घटाता है।

> [!TIP]
> जैसा कि आप देख सकते हैं, Feed Forward network में 3 layers होती हैं। पहली एक linear layer होती है, जो model के अंदर train किए जाने वाले linear weights (parameters) का उपयोग करके dimensions को 4 गुना बढ़ाती है। इसके बाद, उन सभी dimensions पर GELU function का उपयोग करके richer representations को capture करने के लिए non-linear variations लागू किए जाते हैं और अंत में dimensions को मूल आकार में वापस लाने के लिए एक अन्य linear layer का उपयोग किया जाता है।

### **Multi-Head Attention Mechanism**

इसे पहले के section में समझाया जा चुका है।

#### **उद्देश्य और कार्यक्षमता**

- **Multi-Head Self-Attention:** token को encode करते समय model को input sequence के अलग-अलग positions पर ध्यान केंद्रित करने की अनुमति देता है।
- **मुख्य Components:**
- **Queries, Keys, Values:** input के linear projections, जिनका उपयोग attention scores की गणना के लिए किया जाता है।
- **Heads:** parallel रूप से चलने वाले कई attention mechanisms (`num_heads`), जिनमें से प्रत्येक का dimension कम (`head_dim`) होता है।
- **Attention Scores:** queries और keys के dot product के रूप में compute किए जाते हैं, फिर उन्हें scale और mask किया जाता है।
- **Masking:** model को future tokens पर ध्यान देने से रोकने के लिए causal mask लागू किया जाता है (GPT जैसे autoregressive models के लिए महत्वपूर्ण)।
- **Attention Weights:** masked और scaled attention scores का Softmax।
- **Context Vector:** attention weights के अनुसार values का weighted sum।
- **Output Projection:** सभी heads के outputs को combine करने के लिए linear layer।

> [!TIP]
> इस network का उद्देश्य एक ही context में tokens के बीच relations को ढूँढना है। इसके अलावा, overfitting को रोकने के लिए tokens को अलग-अलग heads में विभाजित किया जाता है, हालांकि प्रत्येक head द्वारा खोजे गए अंतिम relations को इस network के अंत में combine किया जाता है।
>
> इसके अलावा, training के दौरान **causal mask** लागू किया जाता है, ताकि किसी token के साथ specific relations देखते समय बाद के tokens को ध्यान में न लिया जाए और **overfitting को रोकने** के लिए कुछ **dropout** भी लागू किया जाता है।

### **Layer** Normalization
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5 # Prevent division by zero during normalization.
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift
```
#### **उद्देश्य और कार्यक्षमता**

- **Layer Normalization:** प्रत्येक batch में हर individual example के features (embedding dimensions) के across inputs को normalize करने के लिए उपयोग की जाने वाली technique।
- **Components:**
- **`eps`:** normalization के दौरान zero से division को रोकने के लिए variance में जोड़ा गया एक छोटा constant (`1e-5`)।
- **`scale` और `shift`:** ये learnable parameters (`nn.Parameter`) हैं, जो model को normalized output को scale और shift करने की अनुमति देते हैं। इन्हें क्रमशः ones और zeros से initialize किया जाता है।
- **Normalization Process:**
- **Mean (`mean`) की गणना:** input `x` का embedding dimension (`dim=-1`) के across mean calculate करता है और broadcasting के लिए dimension को बनाए रखता है (`keepdim=True`)।
- **Variance (`var`) की गणना:** `x` का embedding dimension के across variance calculate करता है और dimension को बनाए रखता है। `unbiased=False` parameter यह सुनिश्चित करता है कि variance की गणना biased estimator का उपयोग करके की जाए (`N-1` के बजाय `N` से divide करके), जो samples के बजाय features पर normalization करते समय उपयुक्त है।
- **Normalize (`norm_x`):** `x` में से mean घटाता है और variance plus `eps` के square root से divide करता है।
- **Scale और Shift:** normalized output पर learnable `scale` और `shift` parameters apply करता है।

> [!TIP]
> लक्ष्य यह सुनिश्चित करना है कि एक ही token के सभी dimensions में mean 0 और variance 1 हो। इसका उद्देश्य **deep neural networks की training को stabilize करना** है, internal covariate shift को कम करके। Internal covariate shift का अर्थ training के दौरान parameters update होने के कारण network activations के distribution में होने वाला बदलाव है।

### **Transformer Block**

_मेट्रिसेस के shapes को बेहतर ढंग से समझने के लिए comments के रूप में जोड़ा गया है:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"]
)
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for attention block
shortcut = x  # shape: (batch_size, seq_len, emb_dim)
x = self.norm1(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.att(x)    # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for feedforward block
shortcut = x       # shape: (batch_size, seq_len, emb_dim)
x = self.norm2(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.ff(x)     # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

return x  # Output shape: (batch_size, seq_len, emb_dim)

```
#### **उद्देश्य और कार्यक्षमता**

- **Layers का संयोजन:** Multi-head attention, feedforward network, layer normalization और residual connections को संयोजित करता है।
- **Layer Normalization:** Stable training के लिए attention और feedforward layers से पहले लागू किया जाता है।
- **Residual Connections (Shortcuts):** Gradient flow को बेहतर बनाने और deep networks की training को सक्षम करने के लिए किसी layer के input को उसके output में जोड़ता है।
- **Dropout:** Regularization के लिए attention और feedforward layers के बाद लागू किया जाता है।

#### **चरण-दर-चरण कार्यक्षमता**

1. **पहला Residual Path (Self-Attention):**
- **Input (`shortcut`):** Residual connection के लिए original input को save करें।
- **Layer Norm (`norm1`):** Input को normalize करें।
- **Multi-Head Attention (`att`):** Self-attention लागू करें।
- **Dropout (`drop_shortcut`):** Regularization के लिए dropout लागू करें।
- **Add Residual (`x + shortcut`):** Original input के साथ संयोजित करें।
2. **दूसरा Residual Path (FeedForward):**
- **Input (`shortcut`):** अगली residual connection के लिए updated input को save करें।
- **Layer Norm (`norm2`):** Input को normalize करें।
- **FeedForward Network (`ff`):** Feedforward transformation लागू करें।
- **Dropout (`drop_shortcut`):** Dropout लागू करें।
- **Add Residual (`x + shortcut`):** पहले residual path के input के साथ संयोजित करें।

> [!TIP]
> Transformer block सभी networks को एक साथ group करता है और training stability तथा results को बेहतर बनाने के लिए कुछ **normalization** और **dropouts** लागू करता है।\
> ध्यान दें कि प्रत्येक network के उपयोग के बाद dropouts किए जाते हैं, जबकि normalization पहले लागू किया जाता है।
>
> इसके अलावा, यह shortcuts का भी उपयोग करता है, जिसमें **किसी network के output को उसके input के साथ जोड़ना** शामिल है। इससे vanishing gradient problem को रोकने में मदद मिलती है, क्योंकि यह सुनिश्चित करता है कि initial layers last layers की तरह "उतना ही" योगदान दें।

### **GPTModel**

_Matrices के shapes को बेहतर ढंग से समझने के लिए comments के रूप में जोड़ा गया है:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
# shape: (vocab_size, emb_dim)

self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
# shape: (context_length, emb_dim)

self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])]
)
# Stack of TransformerBlocks

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(cfg["emb_dim"], cfg["vocab_size"], bias=False)
# shape: (emb_dim, vocab_size)

def forward(self, in_idx):
# in_idx shape: (batch_size, seq_len)
batch_size, seq_len = in_idx.shape

# Token embeddings
tok_embeds = self.tok_emb(in_idx)
# shape: (batch_size, seq_len, emb_dim)

# Positional embeddings
pos_indices = torch.arange(seq_len, device=in_idx.device)
# shape: (seq_len,)
pos_embeds = self.pos_emb(pos_indices)
# shape: (seq_len, emb_dim)

# Add token and positional embeddings
x = tok_embeds + pos_embeds  # Broadcasting over batch dimension
# x shape: (batch_size, seq_len, emb_dim)

x = self.drop_emb(x)  # Dropout applied
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.trf_blocks(x)  # Pass through Transformer blocks
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.final_norm(x)  # Final LayerNorm
# x shape remains: (batch_size, seq_len, emb_dim)

logits = self.out_head(x)  # Project to vocabulary size
# logits shape: (batch_size, seq_len, vocab_size)

return logits  # Output shape: (batch_size, seq_len, vocab_size)
```
#### **उद्देश्य और कार्यक्षमता**

- **Embedding Layers:**
- **Token Embeddings (`tok_emb`):** Token indices को embeddings में परिवर्तित करता है। याद रखें, ये vocabulary में प्रत्येक token के प्रत्येक dimension को दिए गए weights होते हैं।
- **Positional Embeddings (`pos_emb`):** Tokens के क्रम को capture करने के लिए embeddings में positional information जोड़ता है। याद रखें, ये text में token की position के अनुसार उसे दिए गए weights होते हैं।
- **Dropout (`drop_emb`):** Regularisation के लिए embeddings पर लागू किया जाता है।
- **Transformer Blocks (`trf_blocks`):** Embeddings को process करने के लिए `n_layers` transformer blocks का stack।
- **Final Normalization (`final_norm`):** Output layer से पहले layer normalization।
- **Output Layer (`out_head`):** Prediction के लिए final hidden states को vocabulary size में project करके logits उत्पन्न करता है।

> [!TIP]
> इस class का लक्ष्य अन्य सभी उल्लिखित networks का उपयोग करके **एक sequence में अगला token predict करना** है, जो text generation जैसे tasks के लिए fundamental है।
>
> ध्यान दें कि यह **जितने transformer blocks निर्दिष्ट किए गए हैं, उतने का उपयोग करेगा** और प्रत्येक transformer block में एक multi-head attestation net, एक feed forward net और कई normalizations का उपयोग होता है। इसलिए, यदि 12 transformer blocks का उपयोग किया जाता है, तो इसे 12 से multiply करें।
>
> इसके अतिरिक्त, **output** से पहले एक **normalization** layer जोड़ी जाती है और proper dimensions वाले results प्राप्त करने के लिए अंत में एक final linear layer लागू की जाती है। ध्यान दें कि प्रत्येक final vector का size उपयोग की गई vocabulary के बराबर होता है। ऐसा इसलिए है क्योंकि यह vocabulary के अंदर मौजूद प्रत्येक संभावित token के लिए एक probability प्राप्त करने का प्रयास कर रहा है।

## Text Generate करना

पिछले मॉडल की तरह अगले token का prediction करने वाले model के लिए, output से केवल last token की values लेना आवश्यक है (क्योंकि वे predicted token की values होंगी)। ये vocabulary की प्रत्येक entry के लिए एक **value** होंगी। इसके बाद dimensions को ऐसी probabilities में normalize करने के लिए `softmax` function का उपयोग करें जिनका योग 1 हो, और फिर सबसे बड़ी entry का index प्राप्त करें। यही vocabulary में उस word का index होगा।

Code from [https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb](https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb):<sup>[[1]](#references)</sup>
```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
# idx is (batch, n_tokens) array of indices in the current context
for _ in range(max_new_tokens):

# Crop current context if it exceeds the supported context size
# E.g., if LLM supports only 5 tokens, and the context size is 10
# then only the last 5 tokens are used as context
idx_cond = idx[:, -context_size:]

# Get the predictions
with torch.no_grad():
logits = model(idx_cond)

# Focus only on the last time step
# (batch, n_tokens, vocab_size) becomes (batch, vocab_size)
logits = logits[:, -1, :]

# Apply softmax to get probabilities
probas = torch.softmax(logits, dim=-1)  # (batch, vocab_size)

# Get the idx of the vocab entry with the highest probability value
idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # (batch, 1)

# Append sampled index to the running sequence
idx = torch.cat((idx, idx_next), dim=1)  # (batch, n_tokens+1)

return idx


start_context = "Hello, I am"

print("encoded:", encoded)

encoded_tensor = torch.tensor(encoded).unsqueeze(0)
print("encoded_tensor.shape:", encoded_tensor.shape)

model.eval() # disable dropout

out = generate_text_simple(
model=model,
idx=encoded_tensor,
max_new_tokens=6,
context_size=GPT_CONFIG_124M["context_length"]
)

print("Output:", out)
print("Output length:", len(out[0]))
```
