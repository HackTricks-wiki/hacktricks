1. **入力（Tokenized Text）**: プロセスは Tokenized Text から始まり、数値表現に変換されます。
2. **Token Embedding と Positional Embedding Layer**: Tokenized Text は **token embedding** レイヤーと **positional embedding** レイヤーを通過します。これらはシーケンス内の token の位置を捉えるもので、単語の順序を理解するうえで重要です。
3. **Transformer Blocks**: モデルには **12 個の transformer blocks** が含まれており、それぞれに複数のレイヤーがあります。これらの block では、次のシーケンスが繰り返されます:
- **Masked Multi-Head Attention**: モデルが入力テキストの異なる部分に同時に注目できるようにします。
- **Layer Normalization**: training を安定させ、改善するための正規化ステップです。
- **Feed Forward Layer**: attention layer からの情報を処理し、次の token について予測する役割を担います。
- **Dropout Layers**: training 中に unit をランダムに削除することで、過学習を防ぎます。
4. **Final Output Layer**: モデルは **4x50,257 次元の tensor** を出力します。ここで **50,257** は vocabulary のサイズを表します。この tensor の各行は、シーケンス内の次の word を予測するためにモデルが使用する vector に対応します。
5. **Goal**: 目的は、これらの embedding を text に戻すことです。具体的には、出力の最後の行を使用して次の word を生成します。この図では "forward" として表されています。

### コード表現
```python
import torch
import torch.nn as nn
import tiktoken

class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))

class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
return self.layers(x)

class MultiHeadAttention(nn.Module):
def __init__(self, d_in, d_out, context_length, dropout, num_heads, qkv_bias=False):
super().__init__()
assert d_out % num_heads == 0, "d_out must be divisible by num_heads"

self.d_out = d_out
self.num_heads = num_heads
self.head_dim = d_out // num_heads # Reduce the projection dim to match desired output dim

self.W_query = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_key = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_value = nn.Linear(d_in, d_out, bias=qkv_bias)
self.out_proj = nn.Linear(d_out, d_out)  # Linear layer to combine head outputs
self.dropout = nn.Dropout(dropout)
self.register_buffer('mask', torch.triu(torch.ones(context_length, context_length), diagonal=1))

def forward(self, x):
b, num_tokens, d_in = x.shape

keys = self.W_key(x) # Shape: (b, num_tokens, d_out)
queries = self.W_query(x)
values = self.W_value(x)

# We implicitly split the matrix by adding a `num_heads` dimension
# Unroll last dim: (b, num_tokens, d_out) -> (b, num_tokens, num_heads, head_dim)
keys = keys.view(b, num_tokens, self.num_heads, self.head_dim)
values = values.view(b, num_tokens, self.num_heads, self.head_dim)
queries = queries.view(b, num_tokens, self.num_heads, self.head_dim)

# Transpose: (b, num_tokens, num_heads, head_dim) -> (b, num_heads, num_tokens, head_dim)
keys = keys.transpose(1, 2)
queries = queries.transpose(1, 2)
values = values.transpose(1, 2)

# Compute scaled dot-product attention (aka self-attention) with a causal mask
attn_scores = queries @ keys.transpose(2, 3)  # Dot product for each head

# Original mask truncated to the number of tokens and converted to boolean
mask_bool = self.mask.bool()[:num_tokens, :num_tokens]

# Use the mask to fill attention scores
attn_scores.masked_fill_(mask_bool, -torch.inf)

attn_weights = torch.softmax(attn_scores / keys.shape[-1]**0.5, dim=-1)
attn_weights = self.dropout(attn_weights)

# Shape: (b, num_tokens, num_heads, head_dim)
context_vec = (attn_weights @ values).transpose(1, 2)

# Combine heads, where self.d_out = self.num_heads * self.head_dim
context_vec = context_vec.contiguous().view(b, num_tokens, self.d_out)
context_vec = self.out_proj(context_vec) # optional projection

return context_vec

class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"])
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# Shortcut connection for attention block
shortcut = x
x = self.norm1(x)
x = self.att(x)  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

# Shortcut connection for feed forward block
shortcut = x
x = self.norm2(x)
x = self.ff(x)
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

return x


class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])])

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(
cfg["emb_dim"], cfg["vocab_size"], bias=False
)

def forward(self, in_idx):
batch_size, seq_len = in_idx.shape
tok_embeds = self.tok_emb(in_idx)
pos_embeds = self.pos_emb(torch.arange(seq_len, device=in_idx.device))
x = tok_embeds + pos_embeds  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_emb(x)
x = self.trf_blocks(x)
x = self.final_norm(x)
logits = self.out_head(x)
return logits

GPT_CONFIG_124M = {

"n_heads": 12,          # Number of attention heads
"n_layers": 12,         # Number of layers
"drop_rate": 0.1,       # Dropout rate
"qkv_bias": False       # Query-Key-Value bias
}

torch.manual_seed(123)
model = GPTModel(GPT_CONFIG_124M)
out = model(batch)
print("Input batch:\n", batch)
print("\nOutput shape:", out.shape)
print(out)
```

### **GELU活性化関数**
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))
```
#### **目的と機能**

- **GELU (Gaussian Error Linear Unit):** モデルに非線形性を導入する activation function。
- **Smooth Activation:** 負の入力を 0 にする ReLU とは異なり、GELU は入力を出力へ滑らかにマッピングし、負の入力に対しても小さな非ゼロ値を許容します。

### **FeedForward Neural Network**

_行列の shape をより理解しやすくするため、コメントとして追加しています:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

x = self.layers[0](x)# x shape: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[1](x) # x shape remains: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[2](x) # x shape: (batch_size, seq_len, emb_dim)
return x  # Output shape: (batch_size, seq_len, emb_dim)
```
#### **目的と機能**

- **Position-wise FeedForward Network:** 各位置に対して、同一の2層 fully connected network を個別に適用します。
- **Layer の詳細:**
- **First Linear Layer:** 次元数を `emb_dim` から `4 * emb_dim` に拡張します。
- **GELU Activation:** 非線形性を適用します。
- **Second Linear Layer:** 次元数を `emb_dim` に戻します。

> [!TIP]
> ご覧のとおり、Feed Forward network は3つの layer を使用します。最初の layer は、linear weights（model 内で学習する parameters）を使用して次元数を4倍にする linear layer です。次に、すべての次元に GELU function を適用して非線形な変化を加え、より豊かな表現を捉えられるようにします。最後に、もう1つの linear layer を使用して、次元数を元のサイズに戻します。

### **Multi-Head Attention Mechanism**

これは以前のセクションですでに説明しました。

#### **目的と機能**

- **Multi-Head Self-Attention:** token を encoding する際に、model が input sequence 内の異なる位置に注目できるようにします。
- **主要な構成要素:**
- **Queries, Keys, Values:** attention scores の計算に使用される input の linear projections です。
- **Heads:** 並列で動作する複数の attention mechanisms（`num_heads`）で、それぞれ reduced dimension（`head_dim`）を持ちます。
- **Attention Scores:** queries と keys の dot product として計算され、scaling と masking が適用されます。
- **Masking:** model が future tokens に attention することを防ぐため、causal mask が適用されます（GPT のような autoregressive models では重要です）。
- **Attention Weights:** masked かつ scaled された attention scores に対する Softmax の結果です。
- **Context Vector:** attention weights に基づく values の weighted sum です。
- **Output Projection:** すべての heads の outputs を結合する linear layer です。

> [!TIP]
> この network の目的は、同じ context 内にある tokens 間の関係を見つけることです。さらに、overfitting を防ぐために tokens は複数の heads に分割されますが、各 head で見つかった最終的な関係は、この network の最後で結合されます。
>
> さらに、training 中には **causal mask** が適用されるため、ある token との特定の関係を確認する際に後続の tokens が考慮されません。また、**overfitting を防ぐ**ために **dropout** も適用されます。

### **Layer** Normalization
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5 # Prevent division by zero during normalization.
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift
```
#### **目的と機能**

- **Layer Normalization:** バッチ内の各サンプルについて、特徴量（embedding dimensions）全体の入力を正規化するために使用される技術。
- **構成要素:**
- **`eps`:** 正規化中のゼロ除算を防ぐため、分散に加算される小さな定数（`1e-5`）。
- **`scale` と `shift`:** 正規化された出力のスケールとシフトを可能にする学習可能なパラメータ（`nn.Parameter`）。それぞれ 1 と 0 で初期化される。
- **正規化のプロセス:**
- **平均（`mean`）の計算:** embedding dimension（`dim=-1`）全体にわたる入力 `x` の平均を計算し、broadcasting のために次元を維持する（`keepdim=True`）。
- **分散（`var`）の計算:** embedding dimension 全体にわたる `x` の分散を計算し、次元も維持する。`unbiased=False` パラメータにより、分散は不偏推定量（`N-1` で割る）の代わりに、バイアス推定量（`N` で割る）を使用して計算される。これは、サンプルではなく特徴量に対して正規化を行う場合に適している。
- **正規化（`norm_x`）:** `x` から平均を引き、分散に `eps` を加えた値の平方根で割る。
- **スケールとシフト:** 学習可能な `scale` と `shift` パラメータを正規化された出力に適用する。

> [!TIP]
> 目標は、同じ token のすべての次元にわたって、平均を 0、分散を 1 にすることである。これは、**deep neural networks の training を安定させる**ことを目的としている。具体的には、training 中のパラメータ更新によって network activations の分布が変化する internal covariate shift を低減する。

### **Transformer Block**

_行列の shape をより理解しやすくするため、コメントとして追加しています:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"]
)
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for attention block
shortcut = x  # shape: (batch_size, seq_len, emb_dim)
x = self.norm1(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.att(x)    # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for feedforward block
shortcut = x       # shape: (batch_size, seq_len, emb_dim)
x = self.norm2(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.ff(x)     # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

return x  # Output shape: (batch_size, seq_len, emb_dim)

```
#### **目的と機能**

- **レイヤーの構成:** multi-head attention、feedforward network、layer normalization、residual connectionsを組み合わせる。
- **Layer Normalization:** 安定したトレーニングのため、attentionおよびfeedforward layersの前に適用する。
- **Residual Connections (Shortcuts):** レイヤーの入力を出力に加えることで、gradient flowを改善し、deep networksのトレーニングを可能にする。
- **Dropout:** regularizationのため、attentionおよびfeedforward layersの後に適用する。

#### **ステップごとの機能**

1. **最初のResidual Path (Self-Attention):**
- **Input (`shortcut`):** residual connection用に元の入力を保存する。
- **Layer Norm (`norm1`):** 入力を正規化する。
- **Multi-Head Attention (`att`):** self-attentionを適用する。
- **Dropout (`drop_shortcut`):** regularizationのためにdropoutを適用する。
- **Add Residual (`x + shortcut`):** 元の入力と結合する。
2. **2番目のResidual Path (FeedForward):**
- **Input (`shortcut`):** 次のresidual connection用に更新された入力を保存する。
- **Layer Norm (`norm2`):** 入力を正規化する。
- **FeedForward Network (`ff`):** feedforward変換を適用する。
- **Dropout (`drop_shortcut`):** dropoutを適用する。
- **Add Residual (`x + shortcut`):** 最初のresidual pathからの入力と結合する。

> [!TIP]
> transformer blockはすべてのnetworkをまとめ、トレーニングの安定性と結果を改善するために**normalization**と**dropouts**を適用します。\
> 各networkの使用後にdropoutsが実行され、normalizationはその前に適用される点に注目してください。
>
> さらに、networkの出力を入力に**加算する**shortcutsも使用します。これにより、初期のlayersが最後のlayersと「同程度に」貢献することが保証され、vanishing gradient problemの防止に役立ちます。

### **GPTModel**

_行列のshapeをより分かりやすくするため、コメントとして追加しています:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
# shape: (vocab_size, emb_dim)

self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
# shape: (context_length, emb_dim)

self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])]
)
# Stack of TransformerBlocks

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(cfg["emb_dim"], cfg["vocab_size"], bias=False)
# shape: (emb_dim, vocab_size)

def forward(self, in_idx):
# in_idx shape: (batch_size, seq_len)
batch_size, seq_len = in_idx.shape

# Token embeddings
tok_embeds = self.tok_emb(in_idx)
# shape: (batch_size, seq_len, emb_dim)

# Positional embeddings
pos_indices = torch.arange(seq_len, device=in_idx.device)
# shape: (seq_len,)
pos_embeds = self.pos_emb(pos_indices)
# shape: (seq_len, emb_dim)

# Add token and positional embeddings
x = tok_embeds + pos_embeds  # Broadcasting over batch dimension
# x shape: (batch_size, seq_len, emb_dim)

x = self.drop_emb(x)  # Dropout applied
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.trf_blocks(x)  # Pass through Transformer blocks
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.final_norm(x)  # Final LayerNorm
# x shape remains: (batch_size, seq_len, emb_dim)

logits = self.out_head(x)  # Project to vocabulary size
# logits shape: (batch_size, seq_len, vocab_size)

return logits  # Output shape: (batch_size, seq_len, vocab_size)
```
#### **目的と機能**

- **Embedding Layers:**
- **Token Embeddings (`tok_emb`):** トークンインデックスをembeddingに変換します。ここで、語彙内の各トークンの各次元に与えられる重みを指します。
- **Positional Embeddings (`pos_emb`):** トークンの順序を取得するため、embeddingに位置情報を追加します。ここで、テキスト内の位置に応じてトークンに与えられる重みを指します。
- **Dropout (`drop_emb`):** 正則化のためにembeddingに適用されます。
- **Transformer Blocks (`trf_blocks`):** embeddingを処理する`n_layers`個のtransformer blockのstackです。
- **Final Normalization (`final_norm`):** 出力層の前に行うlayer normalizationです。
- **Output Layer (`out_head`):** 最終的なhidden stateを語彙サイズにprojectionし、予測用のlogitを生成します。

> [!TIP]
> このclassの目的は、ここで説明した他のすべてのnetworkを使用して、**sequence内の次のtokenを予測すること**です。これはtext generationなどのtaskの基本となります。
>
> **指定された数のtransformer blockを使用する**点に注目してください。また、各transformer blockは1つのmulti-head attestation net、1つのfeed forward net、複数のnormalizationを使用します。したがって、12個のtransformer blockを使用する場合は、これを12倍します。
>
> さらに、**output**の**前に** **normalization** layerが追加され、最後にfinal linear layerが適用されて、適切な次元の結果が得られます。各final vectorのサイズが使用される語彙のサイズになっている点に注目してください。これは、語彙内の各可能なtokenに対する確率を取得しようとしているためです。

## 学習するパラメータ数

GPT structureが定義されていれば、学習するパラメータ数を求めることができます。<sup>[[1]](#references)</sup>
```python
GPT_CONFIG_124M = {
"vocab_size": 50257,    # Vocabulary size
"context_length": 1024, # Context length
"emb_dim": 768,         # Embedding dimension
"n_heads": 12,          # Number of attention heads

model = GPTModel(GPT_CONFIG_124M)
total_params = sum(p.numel() for p in model.parameters())
print(f"Total number of parameters: {total_params:,}")
# Total number of parameters: 163,009,536
```
### **ステップごとの計算**

#### **1. 埋め込み層：Token Embedding と Position Embedding**

- **Layer:** `nn.Embedding(vocab_size, emb_dim)`
- **Parameters:** `vocab_size * emb_dim`
```python
token_embedding_params = 50257 * 768 = 38,597,376
```
- **層:** `nn.Embedding(context_length, emb_dim)`
- **パラメータ:** `context_length * emb_dim`
```python
position_embedding_params = 1024 * 768 = 786,432
```
**Embeddingパラメータ総数**
```python
embedding_params = token_embedding_params + position_embedding_params
embedding_params = 38,597,376 + 786,432 = 39,383,808
```
#### **2. Transformer ブロック**

Transformer ブロックは12個あるため、1つのブロックのパラメータ数を計算し、その後12倍します。

**Transformer ブロックあたりのパラメータ**

**a. Multi-Head Attention**

- **構成要素:**
- **Query Linear Layer (`W_query`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Key Linear Layer (`W_key`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Value Linear Layer (`W_value`):** `nn.Linear(emb_dim, emb_dim, bias=False)`
- **Output Projection (`out_proj`):** `nn.Linear(emb_dim, emb_dim)`
- **計算:**

- **`W_query`、`W_key`、`W_value` の各層:**

```python
qkv_params = emb_dim * emb_dim = 768 * 768 = 589,824
```

このような層は3つあるため:

```python
total_qkv_params = 3 * qkv_params = 3 * 589,824 = 1,769,472
```

- **Output Projection (`out_proj`):**

```python
out_proj_params = (emb_dim * emb_dim) + emb_dim = (768 * 768) + 768 = 589,824 + 768 = 590,592
```

- **Multi-Head Attention の合計パラメータ:**

```python
mha_params = total_qkv_params + out_proj_params
mha_params = 1,769,472 + 590,592 = 2,360,064
```

**b. FeedForward Network**

- **構成要素:**
- **First Linear Layer:** `nn.Linear(emb_dim, 4 * emb_dim)`
- **Second Linear Layer:** `nn.Linear(4 * emb_dim, emb_dim)`
- **計算:**

- **First Linear Layer:**

```python
ff_first_layer_params = (emb_dim * 4 * emb_dim) + (4 * emb_dim)
ff_first_layer_params = (768 * 3072) + 3072 = 2,359,296 + 3,072 = 2,362,368
```

- **Second Linear Layer:**

```python
ff_second_layer_params = (4 * emb_dim * emb_dim) + emb_dim
ff_second_layer_params = (3072 * 768) + 768 = 2,359,296 + 768 = 2,360,064
```

- **FeedForward の合計パラメータ:**

```python
ff_params = ff_first_layer_params + ff_second_layer_params
ff_params = 2,362,368 + 2,360,064 = 4,722,432
```

**c. Layer Normalizations**

- **構成要素:**
- 各ブロックに `LayerNorm` のインスタンスが2つあります。
- 各 `LayerNorm` には `2 * emb_dim` 個のパラメータ（scale と shift）があります。
- **計算:**

```python
pythonCopy codelayer_norm_params_per_block = 2 * (2 * emb_dim) = 2 * 768 * 2 = 3,072
```

**d. Transformer ブロックあたりの合計パラメータ**
```python
pythonCopy codeparams_per_block = mha_params + ff_params + layer_norm_params_per_block
params_per_block = 2,360,064 + 4,722,432 + 3,072 = 7,085,568
```
**全Transformerブロックの総パラメータ数**
```python
pythonCopy codetotal_transformer_blocks_params = params_per_block * n_layers
total_transformer_blocks_params = 7,085,568 * 12 = 85,026,816
```
#### **3. 最終層**

**a. 最終層の正規化**

- **パラメータ:** `2 * emb_dim`（スケールとシフト）
```python
pythonCopy codefinal_layer_norm_params = 2 * 768 = 1,536
```
**b. 出力射影層 (`out_head`)**

- **層:** `nn.Linear(emb_dim, vocab_size, bias=False)`
- **パラメータ数:** `emb_dim * vocab_size`
```python
pythonCopy codeoutput_projection_params = 768 * 50257 = 38,597,376
```
#### **4. すべてのパラメータを合計**
```python

embedding_params +
total_transformer_blocks_params +
final_layer_norm_params +
output_projection_params
```

## テキストの生成

前述のように次の token を予測するモデルでは、出力から最後の token の値を取得するだけでよい（それらが予測された token の値になるため）。これは**vocabulary の各エントリにつき 1 つの値**となる。その後、`softmax` 関数を使用して各次元を、合計が 1 になる確率へ正規化し、最も大きいエントリの index を取得する。その index が vocabulary 内の単語の index となる。

コードの出典: [https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb](https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb):<sup>[[1]](#references)</sup>
```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
# idx is (batch, n_tokens) array of indices in the current context
for _ in range(max_new_tokens):

# Crop current context if it exceeds the supported context size
# E.g., if LLM supports only 5 tokens, and the context size is 10
# then only the last 5 tokens are used as context
idx_cond = idx[:, -context_size:]

# Get the predictions
with torch.no_grad():
logits = model(idx_cond)

# Focus only on the last time step
# (batch, n_tokens, vocab_size) becomes (batch, vocab_size)
logits = logits[:, -1, :]

# Apply softmax to get probabilities
probas = torch.softmax(logits, dim=-1)  # (batch, vocab_size)

# Get the idx of the vocab entry with the highest probability value
idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # (batch, 1)

# Append sampled index to the running sequence
idx = torch.cat((idx, idx_next), dim=1)  # (batch, n_tokens+1)

return idx


start_context = "Hello, I am"

encoded = tokenizer.encode(start_context)
print("encoded:", encoded)

encoded_tensor = torch.tensor(encoded).unsqueeze(0)
print("encoded_tensor.shape:", encoded_tensor.shape)

model.eval() # disable dropout

out = generate_text_simple(
model=model,
idx=encoded_tensor,
max_new_tokens=6,
context_size=GPT_CONFIG_124M["context_length"]
)

print("Output:", out)
print("Output length:", len(out[0]))
```
