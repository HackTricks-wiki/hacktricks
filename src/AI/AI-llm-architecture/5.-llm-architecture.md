1. **Input (Tokenized Text)**: Mchakato huanza na tokenized text, ambayo hubadilishwa kuwa numerical representations.
2. **Token Embedding and Positional Embedding Layer**: Tokenized text hupitishwa kupitia layer ya **token embedding** na layer ya **positional embedding**, ambayo huhifadhi position ya tokens katika sequence, jambo muhimu kwa kuelewa mpangilio wa maneno.
3. **Transformer Blocks**: Model ina **12 transformer blocks**, kila moja ikiwa na layers nyingi. Blocks hizi hurudia sequence ifuatayo:
- **Masked Multi-Head Attention**: Huwezesha model kuelekeza umakini kwenye sehemu tofauti za input text kwa wakati mmoja.
- **Layer Normalization**: Hatua ya normalization ya kuimarisha na kuboresha training.
- **Feed Forward Layer**: Huwajibika kuchakata taarifa kutoka attention layer na kufanya predictions kuhusu token inayofuata.
- **Dropout Layers**: Layers hizi huzuia overfitting kwa kuondoa units kwa nasibu wakati wa training.
4. **Final Output Layer**: Model hutoa **4x50,257-dimensional tensor**, ambapo **50,257** inawakilisha ukubwa wa vocabulary. Kila row katika tensor hii inalingana na vector ambayo model hutumia kutabiri neno linalofuata katika sequence.
5. **Goal**: Lengo ni kuchukua embeddings hizi na kuzibadilisha tena kuwa text. Hasa, row ya mwisho ya output hutumika kuzalisha neno linalofuata, linalowakilishwa kama "forward" katika diagram hii.

### Code representation
```python
import torch
import torch.nn as nn
import tiktoken

class GELU(nn.Module):
def __init__(self):
super().__init__()

def forward(self, x):
return 0.5 * x * (1 + torch.tanh(
torch.sqrt(torch.tensor(2.0 / torch.pi)) *
(x + 0.044715 * torch.pow(x, 3))
))

class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
return self.layers(x)

class MultiHeadAttention(nn.Module):
def __init__(self, d_in, d_out, context_length, dropout, num_heads, qkv_bias=False):
super().__init__()
assert d_out % num_heads == 0, "d_out must be divisible by num_heads"

self.d_out = d_out
self.num_heads = num_heads
self.head_dim = d_out // num_heads # Reduce the projection dim to match desired output dim

self.W_query = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_key = nn.Linear(d_in, d_out, bias=qkv_bias)
self.W_value = nn.Linear(d_in, d_out, bias=qkv_bias)
self.out_proj = nn.Linear(d_out, d_out)  # Linear layer to combine head outputs
self.dropout = nn.Dropout(dropout)
self.register_buffer('mask', torch.triu(torch.ones(context_length, context_length), diagonal=1))

def forward(self, x):
b, num_tokens, d_in = x.shape

keys = self.W_key(x) # Shape: (b, num_tokens, d_out)
queries = self.W_query(x)
values = self.W_value(x)

# We implicitly split the matrix by adding a `num_heads` dimension
# Unroll last dim: (b, num_tokens, d_out) -> (b, num_tokens, num_heads, head_dim)
keys = keys.view(b, num_tokens, self.num_heads, self.head_dim)
values = values.view(b, num_tokens, self.num_heads, self.head_dim)
queries = queries.view(b, num_tokens, self.num_heads, self.head_dim)

# Transpose: (b, num_tokens, num_heads, head_dim) -> (b, num_heads, num_tokens, head_dim)
keys = keys.transpose(1, 2)
queries = queries.transpose(1, 2)
values = values.transpose(1, 2)

# Compute scaled dot-product attention (aka self-attention) with a causal mask
attn_scores = queries @ keys.transpose(2, 3)  # Dot product for each head

# Original mask truncated to the number of tokens and converted to boolean
mask_bool = self.mask.bool()[:num_tokens, :num_tokens]

# Use the mask to fill attention scores
attn_scores.masked_fill_(mask_bool, -torch.inf)

attn_weights = torch.softmax(attn_scores / keys.shape[-1]**0.5, dim=-1)
attn_weights = self.dropout(attn_weights)

# Shape: (b, num_tokens, num_heads, head_dim)
context_vec = (attn_weights @ values).transpose(1, 2)

# Combine heads, where self.d_out = self.num_heads * self.head_dim
context_vec = context_vec.contiguous().view(b, num_tokens, self.d_out)
context_vec = self.out_proj(context_vec) # optional projection

return context_vec

class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"])
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# Shortcut connection for attention block
shortcut = x
x = self.norm1(x)
x = self.att(x)  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

# Shortcut connection for feed forward block
shortcut = x
x = self.norm2(x)
x = self.ff(x)
x = self.drop_shortcut(x)
x = x + shortcut  # Add the original input back

return x


class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])])

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(
cfg["emb_dim"], cfg["vocab_size"], bias=False
)

def forward(self, in_idx):
batch_size, seq_len = in_idx.shape
tok_embeds = self.tok_emb(in_idx)
pos_embeds = self.pos_emb(torch.arange(seq_len, device=in_idx.device))
x = tok_embeds + pos_embeds  # Shape [batch_size, num_tokens, emb_size]
x = self.drop_emb(x)
x = self.trf_blocks(x)
x = self.final_norm(x)
logits = self.out_head(x)
return logits

GPT_CONFIG_124M = {
"vocab_size": 50257,    # Vocabulary size
"context_length": 1024, # Context length
"emb_dim": 768,         # Embedding dimension
"n_heads": 12,          # Number of attention heads
"n_layers": 12,         # Number of layers
"drop_rate": 0.1,       # Dropout rate
"qkv_bias": False       # Query-Key-Value bias
}

torch.manual_seed(123)
model = GPTModel(GPT_CONFIG_124M)
out = model(batch)
print("Input batch:\n", batch)
print("\nOutput shape:", out.shape)
print(out)
```

### **FeedForward Neural Network**

_Shapes zimeongezwa kama comments ili kuelewa vizuri zaidi shapes za matrices:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class FeedForward(nn.Module):
def __init__(self, cfg):
super().__init__()
self.layers = nn.Sequential(
nn.Linear(cfg["emb_dim"], 4 * cfg["emb_dim"]),
GELU(),
nn.Linear(4 * cfg["emb_dim"], cfg["emb_dim"]),
)

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

x = self.layers[0](x)# x shape: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[1](x) # x shape remains: (batch_size, seq_len, 4 * emb_dim)
x = self.layers[2](x) # x shape: (batch_size, seq_len, emb_dim)
return x  # Output shape: (batch_size, seq_len, emb_dim)
```
#### **Madhumuni na Utendaji**

- **Position-wise FeedForward Network:** Hutumia mtandao wa tabaka mbili uliounganishwa kikamilifu kwa kila position, kwa njia inayojitegemea na inayofanana.
- **Maelezo ya Tabaka:**
- **First Linear Layer:** Hupanua dimensionality kutoka `emb_dim` hadi `4 * emb_dim`.
- **GELU Activation:** Hutumia non-linearity.
- **Second Linear Layer:** Hupunguza dimensionality na kuirudisha hadi `emb_dim`.

> [!TIP]
> Kama unavyoona, Feed Forward network hutumia tabaka 3. La kwanza ni linear layer litakalozidisha dimensions kwa 4 kwa kutumia linear weights (parameters za ku-train ndani ya model). Kisha, function ya GELU hutumiwa katika dimensions zote hizo ili kutumia none-linear variations za kunasa representations zenye utajiri zaidi, na hatimaye linear layer nyingine hutumiwa kurudi kwenye ukubwa wa awali wa dimensions.

### **Multi-Head Attention Mechanism**

Hili tayari lilielezwa katika sehemu iliyotangulia.

#### **Madhumuni na Utendaji**

- **Multi-Head Self-Attention:** Huruhusu model kuzingatia positions tofauti ndani ya input sequence wakati wa ku-encode token.
- **Vipengele Muhimu:**
- **Queries, Keys, Values:** Linear projections za input, zinazotumiwa kukokotoa attention scores.
- **Heads:** Attention mechanisms nyingi zinazoendesha kwa pamoja (`num_heads`), kila moja ikiwa na dimension iliyopunguzwa (`head_dim`).
- **Attention Scores:** Hukokotolewa kama dot product ya queries na keys, kisha huscaled na kumasked.
- **Masking:** Causal mask hutumiwa kuzuia model kuzingatia tokens za baadaye (jambo muhimu kwa autoregressive models kama GPT).
- **Attention Weights:** Softmax ya attention scores zilizomasked na kuscaled.
- **Context Vector:** Jumla yenye uzito ya values, kulingana na attention weights.
- **Output Projection:** Linear layer ya kuunganisha outputs za heads zote.

> [!TIP]
> Lengo la network hii ni kutafuta mahusiano kati ya tokens zilizo katika context moja. Zaidi ya hayo, tokens hugawanywa katika heads tofauti ili kuzuia overfitting, ingawa mahusiano ya mwisho yanayopatikana kwa kila head huunganishwa mwishoni mwa network hii.
>
> Zaidi ya hayo, wakati wa training **causal mask** hutumiwa ili tokens za baadaye zisitiliwe maanani wakati wa kutafuta mahusiano mahususi ya token, na **dropout** fulani pia hutumiwa **kuzuia overfitting**.

### **Layer** Normalization
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class LayerNorm(nn.Module):
def __init__(self, emb_dim):
super().__init__()
self.eps = 1e-5 # Prevent division by zero during normalization.
self.scale = nn.Parameter(torch.ones(emb_dim))
self.shift = nn.Parameter(torch.zeros(emb_dim))

def forward(self, x):
mean = x.mean(dim=-1, keepdim=True)
var = x.var(dim=-1, keepdim=True, unbiased=False)
norm_x = (x - mean) / torch.sqrt(var + self.eps)
return self.scale * norm_x + self.shift
```
#### **Madhumuni na Utendaji**

- **Layer Normalization:** Mbinu inayotumika kusawazisha inputs katika features (embedding dimensions) kwa kila mfano binafsi ndani ya batch.
- **Components:**
- **`eps`:** Constant ndogo (`1e-5`) inayoongezwa kwenye variance ili kuzuia kugawanya kwa sifuri wakati wa normalization.
- **`scale` na `shift`:** Parameters zinazoweza kujifunzwa (`nn.Parameter`) zinazowezesha model kuongeza au kupunguza scale na shift ya output iliyosawazishwa. Huanzishwa kwa ones na zeros, mtawalia.
- **Normalization Process:**
- **Compute Mean (`mean`):** Hukokotoa mean ya input `x` kwenye embedding dimension (`dim=-1`), huku dimension ikihifadhiwa kwa ajili ya broadcasting (`keepdim=True`).
- **Compute Variance (`var`):** Hukokotoa variance ya `x` kwenye embedding dimension, huku pia dimension ikihifadhiwa. Parameter ya `unbiased=False` huhakikisha kuwa variance inakokotolewa kwa kutumia biased estimator (ikigawanywa kwa `N` badala ya `N-1`), jambo linalofaa wakati wa kufanya normalization kwenye features badala ya samples.
- **Normalize (`norm_x`):** Hutoa mean kutoka kwa `x` na kugawanya kwa square root ya variance pamoja na `eps`.
- **Scale and Shift:** Hutumia parameters za `scale` na `shift` zinazoweza kujifunzwa kwenye output iliyosawazishwa.

> [!TIP]
> Lengo ni kuhakikisha mean ya 0 pamoja na variance ya 1 katika dimensions zote za tokeni ileile. Madhumuni ya hili ni **kuimarisha training ya deep neural networks** kwa kupunguza internal covariate shift, ambayo inarejelea mabadiliko katika distribution ya network activations yanayosababishwa na kusasishwa kwa parameters wakati wa training.

### **Transformer Block**

_Shapes zimeongezwa kama comments ili kuelewa vizuri zaidi shapes za matrices:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04

class TransformerBlock(nn.Module):
def __init__(self, cfg):
super().__init__()
self.att = MultiHeadAttention(
d_in=cfg["emb_dim"],
d_out=cfg["emb_dim"],
context_length=cfg["context_length"],
num_heads=cfg["n_heads"],
dropout=cfg["drop_rate"],
qkv_bias=cfg["qkv_bias"]
)
self.ff = FeedForward(cfg)
self.norm1 = LayerNorm(cfg["emb_dim"])
self.norm2 = LayerNorm(cfg["emb_dim"])
self.drop_shortcut = nn.Dropout(cfg["drop_rate"])

def forward(self, x):
# x shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for attention block
shortcut = x  # shape: (batch_size, seq_len, emb_dim)
x = self.norm1(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.att(x)    # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

# Shortcut connection for feedforward block
shortcut = x       # shape: (batch_size, seq_len, emb_dim)
x = self.norm2(x)  # shape remains (batch_size, seq_len, emb_dim)
x = self.ff(x)     # shape: (batch_size, seq_len, emb_dim)
x = self.drop_shortcut(x)  # shape remains (batch_size, seq_len, emb_dim)
x = x + shortcut   # shape: (batch_size, seq_len, emb_dim)

return x  # Output shape: (batch_size, seq_len, emb_dim)

```
#### **Lengo na Utendaji**

- **Muundo wa Tabaka:** Huunganisha multi-head attention, feedforward network, layer normalization, na residual connections.
- **Urekebishaji wa Tabaka:** Hutumika kabla ya attention na feedforward layers kwa mafunzo thabiti.
- **Residual Connections (Shortcuts):** Huongeza ingizo la layer kwenye matokeo yake ili kuboresha mtiririko wa gradient na kuwezesha mafunzo ya networks zenye layers nyingi.
- **Dropout:** Hutumika baada ya attention na feedforward layers kwa ajili ya regularization.

#### **Utendaji Hatua kwa Hatua**

1. **Njia ya Kwanza ya Residual (Self-Attention):**
- **Ingizo (`shortcut`):** Hifadhi ingizo la awali kwa ajili ya residual connection.
- **Layer Norm (`norm1`):** Rekebisha ingizo.
- **Multi-Head Attention (`att`):** Tumia self-attention.
- **Dropout (`drop_shortcut`):** Tumia dropout kwa ajili ya regularization.
- **Ongeza Residual (`x + shortcut`):** Unganisha na ingizo la awali.
2. **Njia ya Pili ya Residual (FeedForward):**
- **Ingizo (`shortcut`):** Hifadhi ingizo lililosasishwa kwa ajili ya residual connection inayofuata.
- **Layer Norm (`norm2`):** Rekebisha ingizo.
- **FeedForward Network (`ff`):** Tumia mabadiliko ya feedforward.
- **Dropout (`drop_shortcut`):** Tumia dropout.
- **Ongeza Residual (`x + shortcut`):** Unganisha na ingizo kutoka kwenye njia ya kwanza ya residual.

> [!TIP]
> Transformer block huunganisha networks zote pamoja na kutumia **normalization** na **dropouts** ili kuboresha uthabiti wa mafunzo na matokeo.\
> Zingatia kwamba dropouts hutumika baada ya kila network kutumika, huku normalization ikitumika kabla.
>
> Zaidi ya hayo, hutumia shortcuts zinazojumuisha **kuongeza matokeo ya network kwenye ingizo lake**. Hii husaidia kuzuia tatizo la vanishing gradient kwa kuhakikisha kwamba layers za mwanzo zinachangia "kwa kiwango sawa" na zile za mwisho.

### **GPTModel**

_Shapes zimeongezwa kama maelezo ili kuelewa vizuri zaidi shapes za matrices:_
```python
# From https://github.com/rasbt/LLMs-from-scratch/tree/main/ch04
class GPTModel(nn.Module):
def __init__(self, cfg):
super().__init__()
self.tok_emb = nn.Embedding(cfg["vocab_size"], cfg["emb_dim"])
# shape: (vocab_size, emb_dim)

self.pos_emb = nn.Embedding(cfg["context_length"], cfg["emb_dim"])
# shape: (context_length, emb_dim)

self.drop_emb = nn.Dropout(cfg["drop_rate"])

self.trf_blocks = nn.Sequential(
*[TransformerBlock(cfg) for _ in range(cfg["n_layers"])]
)
# Stack of TransformerBlocks

self.final_norm = LayerNorm(cfg["emb_dim"])
self.out_head = nn.Linear(cfg["emb_dim"], cfg["vocab_size"], bias=False)
# shape: (emb_dim, vocab_size)

def forward(self, in_idx):
# in_idx shape: (batch_size, seq_len)
batch_size, seq_len = in_idx.shape

# Token embeddings
tok_embeds = self.tok_emb(in_idx)
# shape: (batch_size, seq_len, emb_dim)

# Positional embeddings
pos_indices = torch.arange(seq_len, device=in_idx.device)
# shape: (seq_len,)
pos_embeds = self.pos_emb(pos_indices)
# shape: (seq_len, emb_dim)

# Add token and positional embeddings
x = tok_embeds + pos_embeds  # Broadcasting over batch dimension
# x shape: (batch_size, seq_len, emb_dim)

x = self.drop_emb(x)  # Dropout applied
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.trf_blocks(x)  # Pass through Transformer blocks
# x shape remains: (batch_size, seq_len, emb_dim)

x = self.final_norm(x)  # Final LayerNorm
# x shape remains: (batch_size, seq_len, emb_dim)

logits = self.out_head(x)  # Project to vocabulary size
# logits shape: (batch_size, seq_len, vocab_size)

return logits  # Output shape: (batch_size, seq_len, vocab_size)
```
#### **Madhumuni na Utendaji**

- **Embedding Layers:**
- **Token Embeddings (`tok_emb`):** Hubadilisha token indices kuwa embeddings. Kama ukumbusho, hizi ni weights zinazotolewa kwa kila dimension ya kila tokeni kwenye vocabulary.
- **Positional Embeddings (`pos_emb`):** Huongeza taarifa za nafasi kwenye embeddings ili kunasa mpangilio wa tokeni. Kama ukumbusho, hizi ni weights zinazotolewa kwa tokeni kulingana na nafasi yake kwenye maandishi.
- **Dropout (`drop_emb`):** Hutumika kwenye embeddings kwa ajili ya regularisation.
- **Transformer Blocks (`trf_blocks`):** Stack ya `n_layers` transformer blocks za kuchakata embeddings.
- **Final Normalization (`final_norm`):** Layer normalization kabla ya output layer.
- **Output Layer (`out_head`):** Huprojekta final hidden states kwenye ukubwa wa vocabulary ili kutoa logits za prediction.

> [!TIP]
> Lengo la class hii ni kutumia networks nyingine zote zilizotajwa ili **kutabiri tokeni inayofuata katika sequence**, jambo ambalo ni la msingi kwa tasks kama text generation.
>
> Zingatia kwamba **itatumia transformer blocks nyingi kama ilivyoonyeshwa** na kwamba kila transformer block inatumia multi-head attestation net moja, feed forward net moja na normalizations kadhaa. Kwa hiyo, ikiwa transformer blocks 12 zitatumika, zidisha hii kwa 12.
>
> Zaidi ya hayo, **normalization** layer huongezwa **kabla ya** **output**, na linear layer ya mwisho hutumika mwishoni ili kupata matokeo yenye dimensions sahihi. Zingatia kwamba kila final vector ina ukubwa wa vocabulary iliyotumika. Hii ni kwa sababu inajaribu kupata probability kwa kila tokeni inayowezekana ndani ya vocabulary.

## Generate Text

Kuwa na model inayotabiri token inayofuata kama ile iliyotangulia, kinachohitajika ni kuchukua thamani za token ya mwisho kutoka kwenye output (kwa kuwa zitakuwa za token iliyotabiriwa), ambazo zitakuwa **thamani moja kwa kila ingizo kwenye vocabulary**, kisha kutumia function ya `softmax` kurekebisha dimensions kuwa probabilities ambazo jumla yake ni 1, na hatimaye kupata index ya ingizo kubwa zaidi, ambayo itakuwa index ya neno ndani ya vocabulary.

Code kutoka [https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb](https://github.com/rasbt/LLMs-from-scratch/blob/main/ch04/01_main-chapter-code/ch04.ipynb):<sup>[[1]](#references)</sup>
```python
def generate_text_simple(model, idx, max_new_tokens, context_size):
# idx is (batch, n_tokens) array of indices in the current context
for _ in range(max_new_tokens):

# Crop current context if it exceeds the supported context size
# E.g., if LLM supports only 5 tokens, and the context size is 10
# then only the last 5 tokens are used as context
idx_cond = idx[:, -context_size:]

# Get the predictions
with torch.no_grad():
logits = model(idx_cond)

# Focus only on the last time step
# (batch, n_tokens, vocab_size) becomes (batch, vocab_size)
logits = logits[:, -1, :]

# Apply softmax to get probabilities
probas = torch.softmax(logits, dim=-1)  # (batch, vocab_size)

# Get the idx of the vocab entry with the highest probability value
idx_next = torch.argmax(probas, dim=-1, keepdim=True)  # (batch, 1)

# Append sampled index to the running sequence
idx = torch.cat((idx, idx_next), dim=1)  # (batch, n_tokens+1)

return idx


start_context = "Hello, I am"

encoded = tokenizer.encode(start_context)
print("encoded:", encoded)

encoded_tensor = torch.tensor(encoded).unsqueeze(0)
print("encoded_tensor.shape:", encoded_tensor.shape)

model.eval() # disable dropout

out = generate_text_simple(
model=model,
idx=encoded_tensor,
max_new_tokens=6,
context_size=GPT_CONFIG_124M["context_length"]
)

print("Output:", out)
print("Output length:", len(out[0]))
```
